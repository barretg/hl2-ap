// Half-Life 2 Archipelago. Installed as a Steam sourcemod; `game` is the name
// Steam's library shows. Derived from the retail hl2/gameinfo.txt: the same
// content mounts (plain paths resolve against hl2.exe's folder, the HL2
// install), with this folder as the mod and its bin/ searched first, so our
// server.dll loads and everything else comes from retail HL2.
"GameInfo"
{
	game 		"Half-Life 2 Archipelago"
	title 		"HALF-LIFE'"

	type		singleplayer_only

	gamelogo 0

	FileSystem
	{
		SteamAppId				220

		SearchPaths
		{
			game+mod			hl2/custom/*

			game_lv				hl2/hl2_lv.vpk
			game+mod			hl2/hl2_sound_vo_english.vpk
			game+mod			hl2/hl2_pak.vpk
			game				|all_source_engine_paths|hl2/hl2_textures.vpk
			game				|all_source_engine_paths|hl2/hl2_sound_misc.vpk
			game				|all_source_engine_paths|hl2/hl2_misc.vpk
			platform			|all_source_engine_paths|platform/platform_misc.vpk

			// This folder: the mod path, and where saves and config are written.
			mod+mod_write+default_write_path		|gameinfo_path|.
			game				|gameinfo_path|.

			// Our server.dll first; client.dll and gamepadui.dll fall through
			// to the retail HL2 ones.
			gamebin				|gameinfo_path|bin
			gamebin				hl2/bin

			// Retail hl2's loose files. A mod path as well as game, because
			// for retail HL2 this folder is the mod: GameUI reads its menu
			// files and cfg/chapter*.cfg (the New Game list) from MOD.
			game+mod			|all_source_engine_paths|hl2
			platform			|all_source_engine_paths|platform
		}
	}
}
